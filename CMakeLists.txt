cmake_minimum_required(VERSION 3.21)
project(OrdWeave VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
foreach(option TESTS EXAMPLES BENCHMARKS CAPACITY_SCREEN)
    set(LKS_BUILD_${option} OFF CACHE BOOL "Disabled for embedded dependency" FORCE)
endforeach()
set(LKS_INSTALL OFF CACHE BOOL "Disable dependency installation" FORCE)
add_subdirectory(third_party/LayerKeySort EXCLUDE_FROM_ALL)
add_library(ordweave STATIC src/ordweave.c src/lks_adapter.c src/persistence.c src/wal_format.c src/wal_windows.c src/sha256.c)
add_library(OrdWeave::ordweave ALIAS ordweave)
target_include_directories(ordweave PUBLIC include)
target_compile_features(ordweave PUBLIC c_std_17)
if(WIN32)
    target_link_libraries(ordweave PRIVATE bcrypt)
else()
    message(FATAL_ERROR "Phase 3 backend requires Windows")
endif()
target_link_libraries(ordweave PRIVATE LayerKeySort::layerkeysort)
include(CTest)
if(BUILD_TESTING)
    add_executable(ordweave_smoke tests/smoke.c)
    target_link_libraries(ordweave_smoke PRIVATE OrdWeave::ordweave LayerKeySort::layerkeysort)
    add_test(NAME ordweave_smoke COMMAND ordweave_smoke)
    add_executable(ordweave_core tests/core.c)
    target_link_libraries(ordweave_core PRIVATE OrdWeave::ordweave)
    add_test(NAME ordweave_core COMMAND ordweave_core)
    add_executable(ordweave_failures tests/failures.c src/lks_adapter.c)
    target_include_directories(ordweave_failures PRIVATE include)
    target_link_libraries(ordweave_failures PRIVATE LayerKeySort::layerkeysort)
    add_test(NAME ordweave_failures COMMAND ordweave_failures)
endif()
if(BUILD_TESTING)
    add_library(ordweave_test_storage STATIC src/ordweave.c src/lks_adapter.c src/persistence.c src/wal_format.c src/wal_windows.c src/sha256.c)
    target_include_directories(ordweave_test_storage PUBLIC include PRIVATE src)
    target_compile_definitions(ordweave_test_storage PRIVATE OW_TESTING)
    target_link_libraries(ordweave_test_storage PRIVATE LayerKeySort::layerkeysort bcrypt)
    add_executable(ordweave_persistence tests/persistence.c)
    target_include_directories(ordweave_persistence PRIVATE src)
    target_compile_definitions(ordweave_persistence PRIVATE OW_TESTING)
    target_link_libraries(ordweave_persistence PRIVATE ordweave_test_storage)
    add_test(NAME ordweave_persistence COMMAND ordweave_persistence)
endif()
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME ordweave_format COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/format_oracle.py" $<TARGET_FILE:ordweave_persistence> "${CMAKE_CURRENT_SOURCE_DIR}/tests/format_vectors.json")
endif()
