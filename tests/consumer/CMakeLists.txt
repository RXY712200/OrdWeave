cmake_minimum_required(VERSION 3.21)
project(OrdWeaveConsumer LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(BUILD_TESTING OFF CACHE BOOL "Consumer only" FORCE)
add_subdirectory("${ORDWEAVE_SOURCE_DIR}" ordweave)
add_executable(consumer main.c)
target_link_libraries(consumer PRIVATE OrdWeave::ordweave)
enable_testing()
add_test(NAME consumer_lifecycle COMMAND consumer)
