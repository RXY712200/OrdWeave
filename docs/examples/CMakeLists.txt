cmake_minimum_required(VERSION 3.21)
project(OrdWeaveDocumentationExamples LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT ORDWEAVE_SOURCE_DIR)
    get_filename_component(ORDWEAVE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
set(BUILD_TESTING OFF CACHE BOOL "Embed library without production tests" FORCE)
add_subdirectory("${ORDWEAVE_SOURCE_DIR}" ordweave)
foreach(example quickstart lifecycle)
    add_executable(${example} ${example}.c)
    target_link_libraries(${example} PRIVATE OrdWeave::ordweave)
endforeach()
enable_testing()
add_test(NAME docs_quickstart COMMAND quickstart)
# The lifecycle executable accepts two nonexistent directories with existing parents.
add_test(NAME docs_lifecycle COMMAND lifecycle "${CMAKE_CURRENT_BINARY_DIR}/new-db" "${CMAKE_CURRENT_BINARY_DIR}/new-backup")

if(DOCS_README_SOURCE)
    add_executable(readme_example "${DOCS_README_SOURCE}")
    target_link_libraries(readme_example PRIVATE OrdWeave::ordweave)
    add_test(NAME docs_readme COMMAND readme_example)
endif()
